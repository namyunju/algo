'''
원하는 숫자를 만들기 위한 최소 터치 횟수 
각 값을 노드로 관리하고 해당 노드까지 도달하는 데 필요한 최소 터치 횟수
가중치는 모두 양수임
>> 출발점에서 원하는 지점까지 최단 거리 다익스트라.

'''
'''
다음에 누를 것을 하나씩 선택하는 경우 연산이 어려워짐
2 * 인 상태에서 
2 * 3이 될지
2 * 32가 될지
이걸 매번 하나씩 고려해서 계산하기 어려움. 
>> 가능한 모든 숫자를 다 준비해두고 활용. 
어차피 사용 가능한 숫자는 10개 이하. 전부 만들어도 최대 1000개임.

중단 조건
현재 터치 횟수 + 이후 최소 필요 터치 횟수 > M
연산 결과가 0미만 또는 999 초과

'''

import heapq

T = int(input())
for tc in range(1, T+1):
    N, O, M = map(int, input().split())
    able_nums = set(input().split())
    ops = list(map(int, input().split()))
    W = int(input())
    INF = float('inf')
    ans = -1

    only_num = True
    str_W = str(W)
    for s in str_W:
        if s not in able_nums:
            only_num = False
            break

    if only_num:
        ans = len(str_W)
        print(f'#{tc} {ans}')
        continue

    # 생성 가능 숫자 모음
    # 자릿수를 인덱스로 모아두기
    numbers = [[], [], [], []]

    for n in range(1000):
        s = str(n)
        possible = True

        for d in s:
            if d not in able_nums:
                possible = False
                break
        if possible:
            numbers[len(s)].append(n)

    # 각 숫자까지의 최소 터치 횟수
    dist = [INF] * 1000
    pq = []

    # 시작
    # 맨처음 입력은 숫자. 가능한 모든 숫자를 투입.
    # 거리는 숫자의 길이
    # (터치 횟수, 현재 숫자)
    for length in range(1, 4):
        for n in numbers[length]:
            dist[n] = length
            heapq.heappush(pq, (length, n))

    # 최소 터치 횟수부터 확인
    while pq:
        now_d, value = heapq.heappop(pq)

        # 이미 더 짧은 경로가 발견된 경우 제외
        if now_d > dist[value]:
            continue

        if value == W:
            ans = now_d + 1
            break

        # 추가 연산 시 최소 3회의 터치 필요
        # 연산자, 숫자, 등호
        if now_d + 3 > M:
            continue

        # 다음 숫자 최대 길이
        # 세자리거나 남은 터치횟수에 따라 그 이하 (연산자 1, 등호 1)
        max_len = min(3, M - now_d - 2)

        for length in range(1, max_len + 1):
            cost = now_d + 1 + length

            for n in numbers[length]:
                for op in ops:

                    if op == 1:  
                        r = value + n

                    elif op == 2:  
                        r = value - n

                    elif op == 3:  
                        r = value * n

                    else:  
                        if n == 0:
                            continue
                        r = value // n

                    # 계산 범위 검사
                    if not (0 <= r <= 999):
                        continue

                    # 기존 최소 터치 횟수보다 작은 경우만 갱신
                    if cost < dist[r]:
                        dist[r] = cost
                        heapq.heappush(pq, (cost, r))

    print(f'#{tc} {ans}')
