'''
i번째줄
N//2 -i 부터 +i 

N//2 이후 행부터는
N//2 - (N-i) 부터 +(N-i)
'''
T = int(input())
for tc in range(1, T+1):
    N = int(input())
    board = [list(map(int, input())) for _ in range(N)]
    ans = 0
    for i in range(N//2+1):
        ans += sum(board[i][N//2-i:N//2+i+1])
    for i in range(N//2+1, N):
        ans += sum(board[i][N//2-(N-i)+1:N//2+(N-i)])
    print(f'#{tc} {ans}')
