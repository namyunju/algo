#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;

int main(int argc, char** argv)
{
	int T;
	cin>>T;
    for (int tc=1; tc<=T; tc++) {
        int N;
        cin >> N;
        vector<int> nums(N);
        
        for (int i=0; i<N; i++) {
            cin >> nums[i];
        }
        sort(nums.begin(), nums.end());
        cout << "#" << tc << " ";
        for (int i = 0; i < N; i++) {
            cout << nums[i] << " ";
        }
        cout << "\n";
    }
    
    
	return 0;
}
