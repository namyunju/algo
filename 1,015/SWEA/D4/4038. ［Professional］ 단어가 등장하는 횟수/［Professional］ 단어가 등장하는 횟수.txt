T = int(input())

for tc in range(1, T + 1):
    book = input().strip()
    word = input().strip()

    word_len = len(word)

    pattern = [0] * word_len
    matched = 0

    for i in range(1, word_len):
        while matched > 0 and word[i] != word[matched]:
            # 직전까지 일치
            matched = pattern[matched - 1]

        if word[i] == word[matched]:
            matched += 1
            pattern[i] = matched

    count = 0
    matched = 0

    for char in book:
        while matched > 0 and char != word[matched]:
            matched = pattern[matched - 1]

        if char == word[matched]:
            matched += 1

        if matched == word_len:
            count += 1
            matched = pattern[matched - 1]

    print(f"#{tc} {count}")
