/*
총점은 중간 3.5 기말 4.5 과제 2

10개 평점
10의 배수 N

K 번째 학생의 학점을 출력
K 번째 학생이 전체에서 몇 번째인지

*/

#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int T;
    cin >> T;
    string grades[10] = {"A+", "A0", "A-", "B+", "B0", "B-", "C+", "C0", "C-", "D0"};
    for (int tc = 1; tc <= T; tc++) {
        int N, K;
        cin >> N >> K;
        
        int scores[100];
        int target = 0;
        
        for (int i = 0; i < N; i++) {
            int a, b, c;
            cin >> a >> b >> c;
            scores[i] = a * 35 + b * 45 + c * 20;
            if (i == K-1) {
                target = scores[i];
            }
        }
        sort(scores, scores+N, greater<int>());
        for (int i = 0; i<N; i++) {
            if (scores[i] == target) {
                cout << "#" << tc << " " << grades[i / (N / 10)] << "\n";
                break;
            }
        }
        
    }
    
    return 0;
}
